cmake_minimum_required(VERSION 3.10)
project(JamoJamo)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# vcpkg toolchain 자동 감지
if(DEFINED ENV{VCPKG_ROOT})
    set(CMAKE_TOOLCHAIN_FILE "$ENV{VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake" CACHE STRING "")
else()
    # 일반적인 vcpkg 설치 경로들 시도
    if(EXISTS "C:/vcpkg/scripts/buildsystems/vcpkg.cmake")
        set(CMAKE_TOOLCHAIN_FILE "C:/vcpkg/scripts/buildsystems/vcpkg.cmake" CACHE STRING "")
    elseif(EXISTS "C:/tools/vcpkg/scripts/buildsystems/vcpkg.cmake")
        set(CMAKE_TOOLCHAIN_FILE "C:/tools/vcpkg/scripts/buildsystems/vcpkg.cmake" CACHE STRING "")
    elseif(EXISTS "$ENV{ProgramFiles}/vcpkg/scripts/buildsystems/vcpkg.cmake")
        set(CMAKE_TOOLCHAIN_FILE "$ENV{ProgramFiles}/vcpkg/scripts/buildsystems/vcpkg.cmake" CACHE STRING "")
    endif()
endif()

# vcpkg를 통한 Boost 찾기 (vcpkg toolchain이 설정된 경우)
if(CMAKE_TOOLCHAIN_FILE)
    message(STATUS "Using vcpkg toolchain: ${CMAKE_TOOLCHAIN_FILE}")
    find_package(Boost REQUIRED COMPONENTS nowide)
    message(STATUS "Boost Include Dir: ${Boost_INCLUDE_DIRS}")
else()
    # vcpkg toolchain이 없으면 수동으로 찾기
    if(DEFINED ENV{BOOST_ROOT})
        set(BOOST_INCLUDE_DIR "$ENV{BOOST_ROOT}/include")
    else()
        find_path(BOOST_INCLUDE_DIR boost/nowide/convert.hpp)
    endif()
    if(BOOST_INCLUDE_DIR)
        message(STATUS "Found Boost at: ${BOOST_INCLUDE_DIR}")
    else()
        message(FATAL_ERROR "Boost not found! Please set VCPKG_ROOT or BOOST_ROOT environment variable.")
    endif()
endif()

add_library(jamojamo STATIC src/jamojamo.cpp include/jamojamo.h)
target_include_directories(jamojamo PUBLIC include)
if(CMAKE_TOOLCHAIN_FILE)
    target_link_libraries(jamojamo PRIVATE Boost::nowide)
else()
    target_include_directories(jamojamo PRIVATE ${BOOST_INCLUDE_DIR})
endif()

add_executable(JamoJamoExample examples/main.cpp)
target_link_libraries(JamoJamoExample PRIVATE jamojamo)
if(CMAKE_TOOLCHAIN_FILE)
    target_link_libraries(JamoJamoExample PRIVATE Boost::nowide)
else()
    target_include_directories(JamoJamoExample PRIVATE ${BOOST_INCLUDE_DIR})
endif()
